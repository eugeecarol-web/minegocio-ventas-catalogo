function isBusinessDayAR(dateObj) { if (!(dateObj instanceof Date) || isNaN(dateObj.getTime())) return false; const day = dateObj.getDay(); if (day === 0 || day === 6) return false; const yyyy = dateObj.getFullYear(); const mm = String(dateObj.getMonth() + 1).padStart(2, '0'); const dd = String(dateObj.getDate()).padStart(2, '0'); return !ARG_HOLIDAYS.has(`${yyyy}-${mm}-${dd}`); } function addBusinessDaysAR(dateStr, businessDaysToAdd) { if (!dateStr) return ''; const parts = String(dateStr).split('-'); if (parts.length !== 3) return ''; const year = parseInt(parts[0], 10); const month = parseInt(parts[1], 10) - 1; const day = parseInt(parts[2], 10); const current = new Date(year, month, day); if (isNaN(current.getTime())) return ''; let added = 0; let safetyCounter = 0; while (added < businessDaysToAdd && safetyCounter < 365) { current.setDate(current.getDate() + 1); if (isBusinessDayAR(current)) { added++; } safetyCounter++; } const yyyy = current.getFullYear(); const mm = String(current.getMonth() + 1).padStart(2, '0'); const dd = String(current.getDate()).padStart(2, '0'); return `${yyyy}-${mm}-${dd}`; } function getEstimatedArrivalDate(closeDateStr) { if (!closeDateStr) return ''; return addBusinessDaysAR(String(closeDateStr).slice(0, 10), 3); } function getPaymentDueDate(closeDateStr) { if (!closeDateStr) return '';
